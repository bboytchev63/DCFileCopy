# Define a variable
$day = "Wednesday"

# Use a switch statement to handle different cases
switch ($day) {
    "Monday" {
        Write-Output "Start of the work week!"
    }
    "Tuesday" {
        Write-Output "Second day of the work week."
    }
    "Wednesday" {
        Write-Output "Midweek already!"
    }
    "Thursday" {
        Write-Output "Almost there!"
    }
    "Friday" {
        Write-Output "Last day of the work week!"
    }
    "Saturday" {
        Write-Output "Weekend time!"
    }
    "Sunday" {
        Write-Output "Rest and recharge for the week ahead."
    }
    Default {
        Write-Output "Not a valid day of the week."
    }
}
